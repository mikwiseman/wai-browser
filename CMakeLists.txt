cmake_minimum_required(VERSION 3.21)
project(WaiBrowser VERSION 0.2.0 LANGUAGES C CXX OBJC OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "This first release targets macOS Apple Silicon")
endif()
set(CMAKE_CXX_STANDARD 20)
set(PROJECT_ARCH arm64)
set(WAI_BUILD_NUMBER "1" CACHE STRING "Monotonic Sparkle build number")
set(WAI_UPDATE_FEED "" CACHE STRING "Published HTTPS Sparkle feed")
if(NOT CEF_ROOT)
  file(GLOB CEF_DISTRIBUTIONS "${CMAKE_SOURCE_DIR}/work/cef_binary_*_macosarm64_minimal")
  list(GET CEF_DISTRIBUTIONS 0 CEF_ROOT)
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
include_directories("${CEF_ROOT}" "${CMAKE_SOURCE_DIR}/src")
file(READ "${CMAKE_SOURCE_DIR}/resources/sparkle-public-key.txt" SPARKLE_PUBLIC_KEY)
string(STRIP "${SPARKLE_PUBLIC_KEY}" SPARKLE_PUBLIC_KEY)
set(SPARKLE_ROOT "${CMAKE_SOURCE_DIR}/work/sparkle")
find_library(SPARKLE_LIBRARY Sparkle PATHS "${SPARKLE_ROOT}" NO_DEFAULT_PATH REQUIRED)
configure_file(resources/Info.plist.in "${CMAKE_BINARY_DIR}/Info.plist" @ONLY)
add_executable(wai_browser MACOSX_BUNDLE src/main.mm src/trust.mm shared/WAISite.m)
SET_EXECUTABLE_TARGET_PROPERTIES(wai_browser)
target_compile_options(wai_browser PRIVATE -fobjc-arc "-F${SPARKLE_ROOT}")
target_link_libraries(wai_browser libcef_dll_wrapper ${CEF_STANDARD_LIBS} sqlite3 "${SPARKLE_LIBRARY}")
set_target_properties(wai_browser PROPERTIES OUTPUT_NAME "Wai Browser" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_BINARY_DIR}/Info.plist" BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "@executable_path/../Frameworks" INSTALL_RPATH "@executable_path/../Frameworks")
set(CEF_APP "${CMAKE_BINARY_DIR}/Wai Browser.app")
add_custom_command(TARGET wai_browser PRE_BUILD COMMAND "${CMAKE_COMMAND}" -E rm -rf "${CEF_APP}/Contents/Frameworks/Chromium Embedded Framework.framework" VERBATIM)
COPY_MAC_FRAMEWORK(wai_browser "${CEF_ROOT}/Release" "${CEF_APP}")
foreach(suffix ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" values "${suffix}")
  list(GET values 0 name_suffix)
  list(GET values 1 target_suffix)
  list(GET values 2 HELPER_ID_SUFFIX)
  set(HELPER_NAME "Wai Browser Helper${name_suffix}")
  set(helper_target "wai_helper${target_suffix}")
  configure_file(resources/helper-Info.plist.in "${CMAKE_BINARY_DIR}/${helper_target}.plist" @ONLY)
  add_executable(${helper_target} MACOSX_BUNDLE src/helper.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${helper_target})
  target_link_libraries(${helper_target} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(${helper_target} PROPERTIES OUTPUT_NAME "${HELPER_NAME}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_BINARY_DIR}/${helper_target}.plist")
  add_dependencies(wai_browser ${helper_target})
  add_custom_command(TARGET wai_browser POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_BINARY_DIR}/${HELPER_NAME}.app" "${CEF_APP}/Contents/Frameworks/${HELPER_NAME}.app" VERBATIM)
endforeach()
add_custom_command(TARGET wai_browser POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E rm -rf "${CEF_APP}/Contents/Frameworks/Sparkle.framework"
  COMMAND /usr/bin/ditto "${SPARKLE_ROOT}/Sparkle.framework" "${CEF_APP}/Contents/Frameworks/Sparkle.framework"
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_SOURCE_DIR}/resources/certificates" "${CEF_APP}/Contents/Resources/certificates"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/resources/AppIcon.icns" "${CEF_APP}/Contents/Resources/AppIcon.icns"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/resources/start.html" "${CEF_APP}/Contents/Resources/start.html"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/resources/public_suffix_list.dat" "${CEF_APP}/Contents/Resources/public_suffix_list.dat"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CEF_ROOT}/LICENSE.txt" "${CEF_APP}/Contents/Resources/CEF-LICENSE.txt"
  COMMAND "${CMAKE_COMMAND}" -E copy "${SPARKLE_ROOT}/LICENSE" "${CEF_APP}/Contents/Resources/Sparkle-LICENSE.txt"
  COMMAND "${CMAKE_COMMAND}" -E copy "${CEF_ROOT}/CREDITS.html" "${CEF_APP}/Contents/Resources/Chromium-CREDITS.html"
  VERBATIM
)
enable_testing()
add_executable(trust_test tests/trust_test.mm src/trust.mm)
target_compile_options(trust_test PRIVATE -fobjc-arc)
target_link_libraries(trust_test "-framework Foundation" sqlite3)
add_test(NAME trust COMMAND trust_test "${CMAKE_SOURCE_DIR}/work/test-certificates")
add_executable(site_test tests/site_test.m shared/WAISite.m)
target_compile_options(site_test PRIVATE -fobjc-arc)
target_link_libraries(site_test "-framework Foundation")
add_test(NAME sites COMMAND site_test "${CMAKE_SOURCE_DIR}/resources/public_suffix_list.dat")
